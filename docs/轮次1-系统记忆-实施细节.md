# 第一轮修改：③ 系统记忆（Agent 不写日记 → God 三层压缩）

## 改了什么

- Agent **不再写 Summary （ factual record ）** — Review 阶段只输出 Reflection （个人反思）
- 系统用 **事件日志** 替代 diary ：每次 Activity 结束后记录 `{who, what, with_whom, outcome}`
- God 模型负责 **压缩** ：W6 来时把 W1 压成 ~200 字 L2 要点；W21 来时把一组 L2 再压成 ~50 字 L3 里程碑
- Plan / Review 阶段注入 **分层记忆** 代替旧的 `read_weekly_summaries` ：L1 事件原文（近 5 周） + L2 要点（6-20 周） + L3 里程碑（21 周+）

# ============ Memory System (L1/L2/L3) ============

@property
def event_log(self) -> Path:
    """System event log — not written by agent, written by world.py after activities."""
    return self.generation_dir / "event_log.jsonl"

def write_event_log(self, entry: Dict) -> None:
    """Append one event record. Called from world.py after each activity.
    entry: {time, activity_name, type, with_whom, outcome_summary}
    """
    self._append_jsonl(self.event_log, entry)

def read_event_logs(self, n_weeks: int = 5, exclude_current: bool = True) -> List[Dict]:
    """Read event logs for the past N weeks. L1 memory = full event records."""
    if not self.event_log.exists():
        return []
    return self._read_jsonl(self.event_log, max_weeks=n_weeks, exclude_cur_t=exclude_current)

# L2 memory: ~200-word compressed summaries per week
@property
def l2_memory_dir(self) -> Path:
    return self.memory_dir / "l2"

def save_l2_memory(self, year: int, week: int, compressed: str) -> None:
    """Save a God-compressed L2 summary of one week's events."""
    path = self.l2_memory_dir / f"year={year}" / f"week={week}.json"
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps({"year": year, "week": week, "compressed": compressed}, ensure_ascii=False), encoding="utf-8")

def read_l2_memories(self, from_week: int, to_week: int) -> List[Dict]:
    """Read L2 memories for a given week range. Returns list sorted by time."""
    t = self.clock.get_time()
    year = t.year
    results = []
    for w in range(from_week, to_week + 1):
        path = self.l2_memory_dir / f"year={year}" / f"week={w}.json"
        prev_year_path = self.l2_memory_dir / f"year={year - 1}" / f"week={w}.json"
        for p in [path, prev_year_path]:
            if p.exists():
                results.append(json.loads(p.read_text(encoding="utf-8")))
                break
    return sorted(results, key=lambda x: (x["year"], x["week"]))

# L3 memory: ~50-word milestones for groups of 5 weeks
@property
def l3_memory_dir(self) -> Path:
    return self.memory_dir / "l3"

def save_l3_memory(self, year: int, from_week: int, to_week: int, milestone: str) -> None:
    """Save a God-compressed L3 milestone for a 5-week block."""
    path = self.l3_memory_dir / f"year={year}" / f"w{from_week}-{to_week}.json"
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps({"year": year, "from_week": from_week, "to_week": to_week, "milestone": milestone}, ensure_ascii=False), encoding="utf-8")

def read_l3_memories(self, n_blocks: int = 10) -> List[Dict]:
    """Read L3 milestone records, newest first."""
    if not self.l3_memory_dir.exists():
        return []
    files = sorted(self.l3_memory_dir.rglob("*.json"), reverse=True)
    results = []
    for f in files[:n_blocks]:
        results.append(json.loads(f.read_text(encoding="utf-8")))
    return results
```

### 2B — 新增 `read_layered_memory()` — 三层记忆合并

```
def read_layered_memory(self) -> str:
    """Build the complete 'What You Remember' section for Plan/Review prompts.
    Reads L1 (recent event logs) + L2 (compressed summaries) + L3 (milestones).
    """
    t = self.clock.get_time()
    from src.agents.prompts import MEMORY_CONTEXT
    
    # L1: Recent weeks — full event logs (1-5 weeks)
    events = self.read_event_logs(n_weeks=5, exclude_current=True)
    l1_text = ""
    if events:
        for e in events:
            l1_text += f"- Y{e['year']}-W{e['week']}, {e.get('type','activity')}: {e.get('outcome_summary','')}\n"
    else:
        l1_text = "(No recent events recorded yet.)"
    
    # L2: Fading memories — compressed summaries (6-20 weeks)
    n_weeks_per_year = get_config()["world"]["time"]["n_week"]
    l2_start = max(1, t.week - 20)
    l2_end = t.week - 6
    l2_entries = self.read_l2_memories(from_week=l2_start, to_week=l2_end)
    l2_text = ""
    if l2_entries:
        for e in l2_entries:
            l2_text += f"- W{e['week']}: {e['compressed']}\n"
    else:
        l2_text = "(No fading memories yet.)"
    
    # L3: Distant past — milestones only (>20 weeks)
    l3_entries = self.read_l3_memories(n_blocks=10)
    l3_text = ""
    if l3_entries:
        for e in l3_entries:
            l3_text += f"- Y{e['year']} W{e['from_week']}-{e['to_week']}: {e['milestone']}\n"
    else:
        l3_text = "(No distant memories yet.)"
    
    return MEMORY_CONTEXT.format(l1_memory=l1_text, l2_memory=l2_text, l3_memory=l3_text)
```

def compress_l2_to_l3(agent: "RoleAgent", year: int, from_week: int, to_week: int) -> str:
    """Use God-model to compress a 5-week L2 block into a ~50-word L3 milestone."""
    l2_entries = agent.dm.read_l2_memories(from_week=from_week, to_week=to_week)
    if not l2_entries:
        return "No significant memories from this period."
    
    l2_text = "\n".join(
        f"W{e['week']}: {e['compressed']}" for e in l2_entries
    )
    
    prompt = f"""
You are compressing five weeks of a person's life into a distant memory — one that will be 
recalled only as a brief milestone. Write ~50 words. Ask yourself: if this person looked back 
years later, what ONE thing would they remember from this period?

L2 summaries:
{l2_text}

Respond with the milestone only, no JSON, no meta-commentary.
"""
    messages = [{"role": "user", "content": prompt}]
    response = get_response_with_retry(messages, model_type="god")
    if isinstance(response, list):
        return response[-1].get("content", "")[:200]
    return str(response)[:200]
```
